// SINGLETON GLOBAL - Le player ne sera créé qu'une seule fois
class PlayerService {
  static final PlayerService _instance = PlayerService._internal();
  factory PlayerService() => _instance;
  PlayerService._internal();

  final AudioPlayer _audioPlayer = AudioPlayer();
  final StreamController<void> _stateController = StreamController<void>.broadcast();
  
  bool isPlaying = false;
  String currentTrack = 'En cours de chargement...';
  String currentArtist = '';
  String? artworkUrl;
  Timer? _metadataTimer;

  // NOUVEAU : Historique des 5 derniers morceaux
  List<TrackInfo> trackHistory = [];
  String? _lastTrackId;

  Stream<void> get stateStream => _stateController.stream;

  Future<void> initialize() async {
    await _audioPlayer.setReleaseMode(ReleaseMode.loop);
    await _audioPlayer.setVolume(0.7);
    
    _audioPlayer.onPlayerStateChanged.listen((state) {
      isPlaying = state == PlayerState.playing;
      _notifyListeners();
    });

    // NOUVEAU : Gérer la perte de connexion audio
    _audioPlayer.onPlayerComplete.listen((event) {
      print('Audio complete - reconnecting...');
      _reconnect();
    });

    // Démarrer la récupération des métadonnées
    _startMetadataFetching();
  }

  void _startMetadataFetching() {
    _fetchMetadata();
    _metadataTimer = Timer.periodic(const Duration(seconds: 10), (timer) {
      _fetchMetadata();
    });
  }

  Future<void> _fetchMetadata() async {
    try {
      final response = await http.get(
        Uri.parse('https://www.radiojar.com/api/stations/z21997mep5nwv/now_playing/'),
      );

      if (response.statusCode == 200) {
        final data = json.decode(response.body);
        final newTrack = data['title'] ?? 'Titre inconnu';
        final newArtist = data['artist'] ?? '';
        final newArtwork = data['thumb'];
        
        final trackId = '$newTrack-$newArtist';
        
        if (_lastTrackId != null && _lastTrackId != trackId) {
          _addToHistory(currentTrack, currentArtist, artworkUrl);
        }
        
        _lastTrackId = trackId;
        currentTrack = newTrack;
        currentArtist = newArtist;
        artworkUrl = newArtwork;
        
        _notifyListeners();
      }
    } catch (e) {
      print('Erreur metadata: $e');
    }
  }

  void _addToHistory(String track, String artist, String? artwork) {
    final trackInfo = TrackInfo(
      title: track,
      artist: artist,
      artworkUrl: artwork,
      timestamp: DateTime.now(),
    );
    
    trackHistory.insert(0, trackInfo);
    
    if (trackHistory.length > 5) {
      trackHistory.removeLast();
    }
    
    print('Historique mis à jour : ${trackHistory.length} morceaux');
  }

  // NOUVEAU : Reconnecter automatiquement si perte de flux
  Future<void> _reconnect() async {
    if (isPlaying) {
      print('Reconnecting to stream...');
      try {
        await _audioPlayer.stop();
        await Future.delayed(Duration(milliseconds: 500));
        await _audioPlayer.play(UrlSource('http://stream.radiojar.com/z21997mep5nwv'));
      } catch (e) {
        print('Reconnect error: $e');
        isPlaying = false;
        _notifyListeners();
      }
    }
  }

  Future<void> togglePlayPause() async {
    try {
      if (isPlaying) {
        await _audioPlayer.pause();
        isPlaying = false;
      } else {
        // NOUVEAU : Toujours forcer un nouveau play pour éviter les états incohérents
        await _audioPlayer.stop();
        await _audioPlayer.play(UrlSource('http://stream.radiojar.com/z21997mep5nwv'));
        isPlaying = true;
      }
      _notifyListeners();
    } catch (e) {
      print('Toggle play/pause error: $e');
      isPlaying = false;
      _notifyListeners();
    }
  }

  // NOUVEAU : Méthode pour réinitialiser l'état si besoin
  Future<void> forceResume() async {
    final currentState = await _audioPlayer.state;
    print('Current player state: $currentState');
    
    if (isPlaying && currentState != PlayerState.playing) {
      print('State mismatch detected - forcing reconnect');
      await _reconnect();
    }
  }

  void _notifyListeners() {
    _stateController.add(null);
  }

  void dispose() {
    _metadataTimer?.cancel();
    _audioPlayer.dispose();
    _stateController.close();
  }
}

// Classe TrackInfo - Info d'un morceau
class TrackInfo {
  final String title;
  final String artist;
  final String? artworkUrl;
  final DateTime timestamp;

  TrackInfo({
    required this.title,
    required this.artist,
    this.artworkUrl,
    required this.timestamp,
  });
}

class HomePage extends StatefulWidget {
  const HomePage({Key? key}) : super(key: key);

  @override
  State<HomePage> createState() => _HomePageState();
}

class _HomePageState extends State<HomePage> with WidgetsBindingObserver {
  final PlayerService _playerService = PlayerService();
  final PageController _pageController = PageController();  // ← NOUVEAU
  int _selectedIndex = 0;

  @override
  void initState() {
    super.initState();
    _playerService.initialize();
    WidgetsBinding.instance.addObserver(this);
  }

  @override
  void dispose() {
    _pageController.dispose();  // ← NOUVEAU
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    super.didChangeAppLifecycleState(state);
    
    if (state == AppLifecycleState.resumed) {
      print('App resumed - checking player state');
      _playerService.forceResume();
    }
  }

  void _onItemTapped(int index) {
    setState(() {
      _selectedIndex = index;
    });
    _pageController.animateToPage(  // ← NOUVEAU
      index,
      duration: Duration(milliseconds: 300),
      curve: Curves.easeInOut,
    );
  }

  void _onPageChanged(int index) {  // ← NOUVEAU
    setState(() {
      _selectedIndex = index;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: PageView(  // ← REMPLACÉ : _pages[_selectedIndex] par PageView
        controller: _pageController,
        onPageChanged: _onPageChanged,
        children: const [
          PlayerPage(),
          HistoryPage(),
          ProgramPage(),
          AboutPage(),
        ],
      ),
      bottomNavigationBar: Container(
        decoration: BoxDecoration(
          border: Border(
            top: BorderSide(
              color: const Color(0xFFE8943A).withOpacity(0.12),
              width: 1,
            ),
          ),
        ),
        child: BottomNavigationBar(
          items: const <BottomNavigationBarItem>[
            BottomNavigationBarItem(
              icon: Icon(Icons.radio),
              label: 'Écouter',
            ),
            BottomNavigationBarItem(
              icon: Icon(Icons.history),
              label: 'Historique',
            ),
            BottomNavigationBarItem(
              icon: Icon(Icons.schedule),
              label: 'Programme',
            ),
            BottomNavigationBarItem(
              icon: Icon(Icons.info_outline),
              label: 'À propos',
            ),
          ],
          currentIndex: _selectedIndex,
          selectedItemColor: const Color(0xFFE8943A),
          unselectedItemColor: const Color(0xFF7a756e),
          backgroundColor: const Color(0xFF141414),
          onTap: _onItemTapped,
          type: BottomNavigationBarType.fixed,
        ),
      ),
    );
  }
}

                          padding: EdgeInsets.symmetric(horizontal: 16),
                          itemCount: player.trackHistory.length,
                          itemBuilder: (context, index) {
                            final track = player.trackHistory[index];
                            return _buildTrackCard(track, index);
                          },
                        ),
                ),
              ],
            ),
          ),
        );
      },
    );
  }

  Widget _buildTrackCard(TrackInfo track, int index) {
    return Card(
      margin: EdgeInsets.only(bottom: 12),
      color: Color(0xFF1c1c1c),
      child: ListTile(
        leading: Container(
          width: 60,
          height: 60,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(4),
          ),
          child: ClipRRect(
            borderRadius: BorderRadius.circular(4),
            child: track.artworkUrl != null
                ? Image.network(
                    track.artworkUrl!,
                    fit: BoxFit.cover,
                    errorBuilder: (context, error, stackTrace) {
                      return Container(
                        color: Color(0xFF282828),
                        child: Icon(
                          Icons.music_note,
                          color: Color(0xFFE8943A),
                        ),
                      );
                    },
                  )
                : Container(
                    color: Color(0xFF282828),
                    child: Icon(
                      Icons.music_note,
                      color: Color(0xFFE8943A),
                    ),
                  ),
          ),
        ),
        title: Text(
          track.title,
          style: TextStyle(
            color: Colors.white,
            fontWeight: FontWeight.w600,
          ),
          maxLines: 1,
          overflow: TextOverflow.ellipsis,
        ),
        subtitle: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              track.artist,
              style: TextStyle(
                color: Color(0xFFF0AD5E),
                fontSize: 13,
              ),
              maxLines: 1,
              overflow: TextOverflow.ellipsis,
            ),
            SizedBox(height: 4),
            Text(
              _formatTimestamp(track.timestamp),
              style: TextStyle(
                color: Colors.white38,
                fontSize: 11,
              ),
            ),
          ],
        ),
        trailing: Container(
          padding: EdgeInsets.symmetric(horizontal: 10, vertical: 4),
          decoration: BoxDecoration(
            color: Color(0xFFE8943A).withOpacity(0.2),
            borderRadius: BorderRadius.circular(12),
          ),
          child: Text(
            '#${index + 1}',
            style: TextStyle(
              color: Color(0xFFE8943A),
              fontSize: 12,
              fontWeight: FontWeight.bold,
            ),
          ),
        ),
      ),
    );
  }

  String _formatTimestamp(DateTime time) {
    final now = DateTime.now();
    final difference = now.difference(time);
    
    if (difference.inMinutes < 1) {
      return 'À l\'instant';
    } else if (difference.inMinutes < 60) {
      return 'Il y a ${difference.inMinutes} min';
    } else if (difference.inHours < 24) {
      return 'Il y a ${difference.inHours}h';
    } else {
      return '${time.day}/${time.month} ${time.hour}:${time.minute.toString().padLeft(2, '0')}';
    }
  }
